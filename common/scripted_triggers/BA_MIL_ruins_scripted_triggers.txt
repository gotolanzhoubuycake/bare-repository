BARE_MIL_Ruins_Can_Start_Development = {
	custom_override_tooltip = {
		tooltip = MIL_ruins_development_only_one_tt
		NOT = { has_country_flag = MIL_ruins_development_ongoing }
	}
	custom_override_tooltip = {
		tooltip = MIL_ruins_development_enemy_active_tt
		NOT = { has_country_flag = MIL_ruins_enemy_active }
	}
}

		}
	}
}

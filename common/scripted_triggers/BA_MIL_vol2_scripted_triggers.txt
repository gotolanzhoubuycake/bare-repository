# Vol.2公用条件；正式入口由新蓝图完成365天后的事件开放，不随开局开放。
BARE_MIL_vol2_investigation_open = {
	original_tag = MIL
	has_country_flag = MIL_vol2_investigation_started
	NOT = { has_country_flag = MIL_vol2_investigation_locked }
	NOT = { has_country_flag = MIL_vol2_departed }
}
BARE_MIL_vol2_order_available = {
	BARE_MIL_vol2_investigation_open = yes
	has_completed_focus = MIL_vol2_engineering_help
	NOT = { has_country_flag = MIL_vol2_order_active }
	NOT = { has_country_flag = MIL_vol2_order_cooldown }
	NOT = { has_country_flag = MIL_vol2_vanguard_intel }
	NOT = { has_country_flag = MIL_vol2_order_stopped }
}
BARE_MIL_vol2_action_open = {
	has_country_flag = MIL_vol2_departed
	NOT = { has_country_flag = MIL_vol2_result_settled }
}
BARE_MIL_vol2_auto_visible = {
	OR = {
		has_completed_focus = MIL_vol2_rescue_start
		has_country_flag = MIL_vol2_departed
	}
}
BARE_MIL_vol2_normal_success = {
	OR = {
		check_variable = { MIL_vol2_action_time <= 5 }
		check_variable = { MIL_vol2_aid_level >= 2 }
		NOT = { has_country_flag = MIL_vol2_rescue_injury }
		check_variable = { MIL_vol2_progress >= 40 }
	}
}

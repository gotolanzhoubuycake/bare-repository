BARE_is_surrender_trigger = {
	ROOT = {
		if = {
			limit = { is_in_faction = no }
			is_puppet = no
		}
		else = {
			all_allied_country = {
				if = {
					limit = {
						has_war_with = FROM
						is_major = yes
					}
					has_capitulated = yes
				}
			}
		}
	}
}

            has_war_with = CAH
        }

            CAB = {
                has_capitulated = yes
            }
            CAH = {
                has_capitulated = yes
            }
            AND = {
                CAH = {
                    surrender_progress > 0.5
                }
                CAB = {
                    surrender_progress > 0.5
                }

            ALF = {

on_actions = {
    # 2026-05-20: 接入晄轮大祭月度循环；剩余月份归零后进入待结算状态，不再执行月更
    on_monthly_MIL = {
        effect = {
            if = {
                limit = {
                    original_tag = MIL
                    has_global_flag = BARE_MIL_Sport_prepare_flag
                    NOT = { has_country_flag = BARE_MIL_Sport_settlement_pending_flag }
                    check_variable = {
                        BARE_MIL_sport_Month_Time_Var > 0
                    }
                }
                BARE_MIL_Sport_Monthly_effect = yes
            }
            if = {
                limit = {
                    original_tag = MIL
                    has_country_flag = BARE_MIL_Sport_support_normalizing_flag
                }
                # 2026-05-20: 结算后不再跑筹备月更，只保留支持度向50回正的轻量逻辑。
                BARE_MIL_Sport_Post_Settlement_Support_Normalize_effect = yes
            }
        }
    }

    # 千年-凯撒战争每周事件
    # Wiki参考：https://hoi4.paradoxwikis.com/On_actions
    # 用于处理每周减少凯撒经济状况的逻辑
# =====================================================
# 千年-凯撒战争经济状况机制
# Wiki参考：https://hoi4.paradoxwikis.com/On_actions
#
# 可用全局变量（供机制/UI显示使用）：
#   global.cae_economic_situation          - 凯撒经济状况基础值（初始120，受每周规则4-7和决议累积修改）
#   global.cae_final_economic_situation    - 凯撒经济状况最终值（基础值 + 规则1-3快照偏移量，每周结算）
#   global.cae_add_economic_situation      - 规则1-3快照偏移量（每周结算时计算）
#   global.cae_last_add_eco               - 上周规则1-3偏移量（用于计算变化量）
#   global.cae_predicted_weekly_change     - 本周经济状况预测变化量（规则1-7合计，每日更新）
#   global.MIL_cae_war_cae_army_attack_factor       - 凯撒部队攻击修正（每周结算）
#   global.MIL_cae_war_cae_army_defence_factor      - 凯撒部队防御修正（每周结算）
#   global.MIL_cae_war_cae_army_org_factor          - 凯撒部队组织度上限修正（每周结算）
#   global.MIL_cae_war_cae_army_org_regain          - 凯撒部队组织度恢复修正（每周结算）
#   global.MIL_cae_war_cae_air_mission_efficiency   - 凯撒空军效率修正（每周结算）
#   global.cae_recruit_equipment_factor             - 凯撒招募部队初始装备系数（0~1，每周结算，经济<100时递减，每降低1减少0.005，最低为0.5）
# =====================================================

    # 用于处理千年决议每周减少凯撒经济状况的逻辑
    on_weekly_MIL = {
        effect = {
            # 检查是否是千年
            if = {
                limit = {
                    original_tag = MIL
                }

                # 决议1效果：每周凯撒经济状况-1
                if = {
                    limit = {
                        has_country_flag = MIL_cae_eco_weekly_minus_1
                    }
                    subtract_from_variable = {
                        var = global.cae_economic_situation
                        value = 1
                    }
                }

                # 决议2效果：每周凯撒经济状况-1.5（在-1%基础上）
                if = {
                    limit = {
                        has_country_flag = MIL_cae_eco_weekly_minus_1_5
                    }
                    subtract_from_variable = {
                        var = global.cae_economic_situation
                        value = 1.5
                    }
                }
            }
        }
    }

    # =====================================================
    # 千年-凯撒战争每日预测
    # 每日计算规则1-7的预测值，存入全局变量供显示
    # 同时每日更新最终经济状况和战斗力修正
    # 规则1：凯撒每控制一个千年省份，经济状况+2%
    # 规则2：凯撒累计每损失7.5k人力，经济状况-1%
    # 规则3：千年累计每损失2k人力，凯撒经济状况+1%
    # 规则4：凯撒每控制n个千年省份，每周经济状况+0.5n%
    # 规则5：凯撒每比千年多出1k服役人口，每周经济状况+0.1%
    # 规则6：凯撒每拥有10k*n个服役人口，每周经济状况-0.1n%
    # 规则7：凯撒每失去一个核心省份（被千年控制），每周经济状况-10%
    # =====================================================
    on_daily_CAE = {
        effect = {
            if = {
                limit = {
                    has_country_flag = CAE_MIL_war_started
                }

                # =========================================
                # 规则1-3预测：快照偏移量
                # =========================================
                set_temp_variable = { add_eco = 0 }

                # 规则1预测：凯撒每控制一个千年省份，经济状况+2%
                CAE = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = MIL
                        }
                        add_to_temp_variable = { add_eco = 2 }
                    }
                }

                # 规则2预测：凯撒累计每损失7.5k人力，经济状况-1%
                set_temp_variable = { death_population_cae = CAE.casualties_k }
                divide_temp_variable = { death_population_cae = 7.5 }
                subtract_from_temp_variable = { add_eco = death_population_cae }

                # 规则3预测：千年累计每损失2k人力，凯撒经济状况+1%
                set_temp_variable = { death_population_mil = MIL.casualties_k }
                divide_temp_variable = { death_population_mil = 2 }
                add_to_temp_variable = { add_eco = death_population_mil }

                # =========================================
                # 规则4-7预测：每周累积变化量
                # =========================================
                set_temp_variable = { predicted_change = 0 }

                # 规则4预测：凯撒每控制n个千年省份，每周经济状况+0.5n%
                CAE = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = MIL
                        }
                        add_to_temp_variable = { predicted_change = 0.5 }
                    }
                }

                # 规则5预测：凯撒每比千年多出1k服役人口，每周经济状况+0.1%
                set_temp_variable = { manpower_advantage = 0 }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_army_manpower_k }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_navy_manpower_k }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_airforce_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_army_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_navy_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_airforce_manpower_k }
                clamp_temp_variable = {
                    var = manpower_advantage
                    min = 0
                }
                divide_temp_variable = { manpower_advantage = 10 }
                add_to_temp_variable = { predicted_change = manpower_advantage }

                # 规则6预测：凯撒每拥有10k*n个服役人口，每周经济状况-0.1n%
                set_temp_variable = { manpower_cost = 0 }
                add_to_temp_variable = { manpower_cost = CAE.deployed_army_manpower_k }
                add_to_temp_variable = { manpower_cost = CAE.deployed_navy_manpower_k }
                add_to_temp_variable = { manpower_cost = CAE.deployed_airforce_manpower_k }
                divide_temp_variable = { manpower_cost = 100 }
                subtract_from_temp_variable = { predicted_change = manpower_cost }

                # 规则7预测：凯撒每失去一个核心省份（被千年控制），每周经济状况-10%
                MIL = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = CAE
                        }
                        subtract_from_temp_variable = { predicted_change = 10 }
                    }
                }

                # 计算规则1-3本周偏移量变化 = 当前偏移量 - 上周偏移量
                set_temp_variable = { add_eco_change = add_eco }
                subtract_from_temp_variable = { add_eco_change = global.cae_last_add_eco }
                # 合并规则1-3变化量和规则4-7变化量，存入全局变量（仅预测，不结算）
                add_to_temp_variable = { predicted_change = add_eco_change }
                set_variable = { global.cae_predicted_weekly_change = predicted_change }
            }
        }
    }

    # =====================================================
    # 千年-凯撒战争每周经济调整（合并所有规则）
    # 规则1-3：快照偏移量，每次重新计算
    # 规则1：凯撒每控制一个千年省份，经济状况+2%
    # 规则2：凯撒累计每损失7.5k人力，经济状况-1%
    # 规则3：千年累计每损失2k人力，凯撒经济状况+1%
    # 规则4-7：每周累积变化，直接修改基础经济状况
    # 规则4：凯撒每控制n个千年省份，每周经济状况+0.5n%
    # 规则5：凯撒每比千年多出1k服役人口，每周经济状况+0.1%
    # 规则6：凯撒每拥有10k*n个服役人口，每周经济状况-0.1n%
    # 规则7：凯撒每失去一个核心省份（被千年控制），每周经济状况-10%
    # =====================================================
    on_weekly_CAE = {
        effect = {
            if = {
                limit = {
                    has_country_flag = CAE_MIL_war_started
                }

                # =========================================
                # 规则1-3：快照偏移量（基于累积数据，每周重新计算）
                # =========================================
                set_temp_variable = { add_eco = 0 }

                # 规则1：凯撒每控制一个千年省份，经济状况+2%
                CAE = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = MIL
                        }
                        add_to_temp_variable = { add_eco = 2 }
                    }
                }

                # 规则2：凯撒累计每损失7.5k人力，经济状况-1%
                set_temp_variable = { death_population_cae = CAE.casualties_k }
                divide_temp_variable = { death_population_cae = 7.5 }
                subtract_from_temp_variable = { add_eco = death_population_cae }

                # 规则3：千年累计每损失2k人力，凯撒经济状况+1%
                set_temp_variable = { death_population_mil = MIL.casualties_k }
                divide_temp_variable = { death_population_mil = 2 }
                add_to_temp_variable = { add_eco = death_population_mil }

                # =========================================
                # 规则4-7：每周累积变化，直接修改基础经济状况
                # =========================================
                set_temp_variable = { weekly_change = 0 }

                # 规则4：凯撒每控制n个千年省份，每周经济状况+0.5n%
                CAE = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = MIL
                        }
                        add_to_temp_variable = { weekly_change = 0.5 }
                    }
                }

                # 规则5：凯撒每比千年多出1k服役人口，每周经济状况+0.1%
                # 计算公式：(凯撒服役人口 - 千年服役人口) / 10
                set_temp_variable = { manpower_advantage = 0 }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_army_manpower_k }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_navy_manpower_k }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_airforce_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_army_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_navy_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_airforce_manpower_k }
                clamp_temp_variable = {
                    var = manpower_advantage
                    min = 0
                }
                divide_temp_variable = { manpower_advantage = 10 }
                add_to_temp_variable = { weekly_change = manpower_advantage }

                # 规则6：凯撒每拥有10k*n个服役人口，每周经济状况-0.1n%
                # 计算公式：凯撒总服役人口 / 100
                set_temp_variable = { manpower_cost = 0 }
                add_to_temp_variable = { manpower_cost = CAE.deployed_army_manpower_k }
                add_to_temp_variable = { manpower_cost = CAE.deployed_navy_manpower_k }
                add_to_temp_variable = { manpower_cost = CAE.deployed_airforce_manpower_k }
                divide_temp_variable = { manpower_cost = 100 }
                subtract_from_temp_variable = { weekly_change = manpower_cost }

                # 规则7：凯撒每失去一个核心省份（被千年控制），每周经济状况-10%
                MIL = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = CAE
                        }
                        subtract_from_temp_variable = { weekly_change = 10 }
                    }
                }

                # 将每周变化累积到基础经济状况
                add_to_variable = { global.cae_economic_situation = weekly_change }

                # =========================================
                # 计算最终经济状况 = 基础值 + 快照偏移量
                # =========================================
                set_variable = { global.cae_final_economic_situation = global.cae_economic_situation }
                set_variable = { global.cae_add_economic_situation = add_eco }
                add_to_variable = { global.cae_final_economic_situation = global.cae_add_economic_situation }

                # =========================================
                # 战斗力影响计算
                # 当经济状况高于/低于100时，每偏离2点=1%战斗力影响
                # 影响项：部队攻防、组织度上限及恢复速度、空军效率
                # 当经济状况低于100时，每降低2点，新招募部队初始装备值-1
                # =========================================
                set_temp_variable = { MIL_cae_war_temp_calc = global.cae_final_economic_situation }
                subtract_from_temp_variable = { MIL_cae_war_temp_calc = 100 }
                divide_temp_variable = { MIL_cae_war_temp_calc = 200 }

                set_variable = { global.MIL_cae_war_cae_army_attack_factor = MIL_cae_war_temp_calc }
                set_variable = { global.MIL_cae_war_cae_army_defence_factor = MIL_cae_war_temp_calc }
                set_variable = { global.MIL_cae_war_cae_army_org_factor = MIL_cae_war_temp_calc }
                set_variable = { global.MIL_cae_war_cae_army_org_regain = MIL_cae_war_temp_calc }
                set_variable = { global.MIL_cae_war_cae_air_mission_efficiency = MIL_cae_war_temp_calc }

                # 记录本周偏移量，供下周daily计算变化量使用
                set_variable = { global.cae_last_add_eco = add_eco }

                # =========================================
                # 招募部队初始装备值计算
                # 默认值1.0（100%），经济状况低于100时每降低2点减少1%
                # =========================================
                set_temp_variable = { equip_penalty = 100 }
                subtract_from_temp_variable = { equip_penalty = global.cae_final_economic_situation }
                clamp_temp_variable = { var = equip_penalty min = 0 }
                divide_temp_variable = { equip_penalty = 200 }
                set_temp_variable = { equip_factor = 1 }
                subtract_from_temp_variable = { equip_factor = equip_penalty }
                clamp_temp_variable = { var = equip_factor min = 0 max = 1 }
                set_variable = { global.cae_recruit_equipment_factor = equip_factor }

                # =========================================
                # 凯撒经济崩溃判定
                # 当最终经济状况小于0时，凯撒战败
                # =========================================
                if = {
                    limit = {
                        check_variable = { global.cae_final_economic_situation < 0 }
                        NOT = { has_global_flag = CAE_MIL_war_cae_defeated }
                    }
                    set_global_flag = CAE_MIL_war_cae_defeated

                    # 双方白和平（包括参战的凯撒子公司）
                    MIL = {
                        white_peace = CAE
                        white_peace = CAP
                        white_peace = CAB
                        white_peace = CAH
                        white_peace = CAO
                    }

                    # MIL获得民族精神"永远飘扬的光辉旗帜"并依次触发三个战后事件
                    MIL = {
                        add_ideas = MIL_glorious_banner_forever
                        country_event = { id = MIL_cae_war.10 days = 1 }  # 光辉的胜利
                        country_event = { id = MIL_cae_war.11 days = 2 }  # 审判战犯
                        country_event = { id = MIL_cae_war.12 days = 3 }  # 凯撒集团大解体
                    }

                    # CAE获得民族精神"史上最失败的投资"和"崩溃的公司体系"
                    CAE = {
                        add_ideas = CAE_worst_investment_ever
                        add_ideas = CAE_collapsed_corporate_system
                    }

                    # 所有凯撒子公司与凯撒总部解除同盟并独立
                    CAE = {
                        release_puppet = CAP
                        release_puppet = CAB
                        release_puppet = CAH
                        release_puppet = CAO
                    }
                    CAP = { leave_faction = yes }
                    CAB = { leave_faction = yes }
                    CAH = { leave_faction = yes }
                    CAO = { leave_faction = yes }

                    # 战争结束，清除国家旗帜，停止后续每周经济结算
                    CAE = { clr_country_flag = CAE_MIL_war_started }
                    MIL = { clr_country_flag = CAE_MIL_war_started }
                }

                # =========================================
                # 凯撒经济胜利判定
                # 当最终经济状况大于200时，凯撒战胜
                # =========================================
                if = {
                    limit = {
                        check_variable = { global.cae_final_economic_situation > 200 }
                        NOT = { has_global_flag = CAE_MIL_war_cae_victory }
                        NOT = { has_global_flag = CAE_MIL_war_cae_defeated }
                    }
                    set_global_flag = CAE_MIL_war_cae_victory

                    # CAE获取在MIL_cae_war.6占领的所有state
                    CAE = {
                        # 马尔斯大道占领区
                        transfer_state = 31
                        transfer_state = 471
                        transfer_state = 32
                        transfer_state = 730
                        transfer_state = 33
                        transfer_state = 731
                        transfer_state = 473
                        transfer_state = 320
                        # 千年大街
                        transfer_state = 733
                        transfer_state = 321
                        transfer_state = 322
                        transfer_state = 732
                        transfer_state = 323
                        # 其他占领区
                        transfer_state = 333
                        transfer_state = 341
                        transfer_state = 331
                        transfer_state = 477
                        transfer_state = 745
                        transfer_state = 746
                        transfer_state = 476
                    }

                    # 双方白和平（包括参战的凯撒子公司）
                    MIL = {
                        white_peace = CAE
                        white_peace = CAP
                        white_peace = CAB
                        white_peace = CAH
                        white_peace = CAO
                    }

                    # 战争结束，清除国家旗帜，停止后续每周经济结算
                    CAE = { clr_country_flag = CAE_MIL_war_started }
                    MIL = { clr_country_flag = CAE_MIL_war_started }
                }
            }
        }
    }
}

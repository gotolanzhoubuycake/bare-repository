    # 千年-凯撒战争月度事件

# =====================================================
# 千年-凯撒战争经济状况机制
# Wiki参考：https://hoi4.paradoxwikis.com/On_actions
#
# 可用全局变量（供机制/UI显示使用）：
#   global.cae_economic_situation          - 凯撒经济状况基础值（初始80，受月度规则4-7和决议累积修改）
#   global.cae_final_economic_situation    - 凯撒经济状况最终值（基础值 + 规则1-3快照偏移量，每月结算）
#   global.cae_economic_progress_frame     - 决议GUI进度条帧值（最终经济状况0~200换算为0~100）
#   global.cae_add_economic_situation      - 规则1-3快照偏移量（每月结算时计算）
#   global.cae_last_add_eco               - 上月规则1-3偏移量（用于计算变化量）
#   global.cae_predicted_monthly_change    - 本月经济状况预测变化量（规则1-7及千年决议每月扣减合计，每日更新）
#   global.MIL_cae_war_cae_army_attack_factor       - 凯撒部队攻击修正（每月结算）
#   global.MIL_cae_war_cae_army_defence_factor      - 凯撒部队防御修正（每月结算）
#   global.MIL_cae_war_cae_army_org_factor          - 凯撒部队组织度上限修正（每月结算）
#   global.MIL_cae_war_cae_army_org_regain          - 凯撒部队组织度恢复修正（每月结算）
#   global.MIL_cae_war_cae_air_mission_efficiency   - 凯撒空军效率修正（每月结算）
#   global.cae_recruit_equipment_factor             - 凯撒招募部队初始装备系数（0~1，每月结算，经济<100时递减，每降低1减少0.005，最低为0.5）
#
# 凯撒战争决议 GUI 缓存变量（每日刷新，供 localisation 直接显示）：
#   global.MIL_cae_war_GUI_CAE_factory_count               - 凯撒工厂总数（军工 + 民工）
#   global.MIL_cae_war_GUI_CAE_deployed_manpower           - 凯撒服役人数（陆海空 deployed_*_manpower_k 合计，单位为k）
#   global.MIL_cae_war_GUI_CAE_casualties                  - 凯撒总伤亡（casualties_k，单位为k）
#   global.MIL_cae_war_GUI_CAE_occupied_state_count         - 凯撒当前控制的千年核心省份数量
#   global.MIL_cae_war_GUI_CAE_remaining_core_state_count   - 凯撒当前仍控制的凯撒核心省份数量
#   global.MIL_cae_war_GUI_MIL_factory_count               - 千年工厂总数（军工 + 民工）
#   global.MIL_cae_war_GUI_MIL_deployed_manpower           - 千年服役人数（陆海空 deployed_*_manpower_k 合计，单位为k）
#   global.MIL_cae_war_GUI_MIL_casualties                  - 千年总伤亡（casualties_k，单位为k）

#   global.cae_economic_progress_frame                      - UI 竖向进度条帧值，0为蓝色空条，100为黄色满条

    # =====================================================
    # 千年-凯撒战争每日预测
    # 每日计算规则1-7和千年决议每月扣减的预测值，存入全局变量供显示
    # 同时刷新 GUI 缓存；最终经济状况和战斗力修正由 on_monthly_CAE 结算
    # 规则1：凯撒每控制一个千年省份，经济状况+2%
    # 规则2：凯撒累计每损失7.5k人力，经济状况-1%
    # 规则3：千年累计每损失2k人力，凯撒经济状况+1%
    # 规则4：凯撒每控制n个千年省份，每月经济状况+0.5n%
    # 规则5：凯撒每比千年多出1k服役人口，每月经济状况+0.1%
    # 规则6：凯撒每拥有1k个服役人口，每月经济状况-0.01%
    # 规则7：凯撒每失去一个核心省份（被千年控制），每月经济状况-10%
    # =====================================================
    on_daily_CAE = {
        effect = {
            if = {
                limit = {
                    has_country_flag = CAE_MIL_war_started
                }

                # =========================================
                # 决议 GUI 数据缓存
                # 左列：凯撒工厂、服役人数、总伤亡、占据千年省份、剩余凯撒核心省份
                # 右列：千年工厂、服役人数、总伤亡、占据凯撒省份、剩余千年核心省份
                # 注意：MIL_cae_war_GUI_temp_value 是本段复用的临时累加器，每个统计项开始前都会重置。
                # =========================================

                # 工厂数量：直接缓存国家当前军工 + 民工总数，显示为整数。
                CAE = {
                    set_variable = { global.MIL_cae_war_GUI_CAE_factory_count = num_of_military_factories }
                    add_to_variable = { global.MIL_cae_war_GUI_CAE_factory_count = num_of_civilian_factories }
                }
                MIL = {
                    set_variable = { global.MIL_cae_war_GUI_MIL_factory_count = num_of_military_factories }
                    add_to_variable = { global.MIL_cae_war_GUI_MIL_factory_count = num_of_civilian_factories }
                }

                # 凯撒服役人数：deployed_*_manpower_k 的单位是千人，GUI 也按k显示。
                set_temp_variable = { MIL_cae_war_GUI_temp_value = 0 }
                add_to_temp_variable = { MIL_cae_war_GUI_temp_value = CAE.deployed_army_manpower_k }
                add_to_temp_variable = { MIL_cae_war_GUI_temp_value = CAE.deployed_navy_manpower_k }
                add_to_temp_variable = { MIL_cae_war_GUI_temp_value = CAE.deployed_airforce_manpower_k }
                set_variable = { global.MIL_cae_war_GUI_CAE_deployed_manpower = MIL_cae_war_GUI_temp_value }

                # 千年服役人数：同样合计陆海空服役人数，GUI 保持k单位。
                set_temp_variable = { MIL_cae_war_GUI_temp_value = 0 }
                add_to_temp_variable = { MIL_cae_war_GUI_temp_value = MIL.deployed_army_manpower_k }
                add_to_temp_variable = { MIL_cae_war_GUI_temp_value = MIL.deployed_navy_manpower_k }
                add_to_temp_variable = { MIL_cae_war_GUI_temp_value = MIL.deployed_airforce_manpower_k }
                set_variable = { global.MIL_cae_war_GUI_MIL_deployed_manpower = MIL_cae_war_GUI_temp_value }

                # 总伤亡：casualties_k 的单位也是千人，GUI 也按k显示。
                set_temp_variable = { MIL_cae_war_GUI_temp_value = CAE.casualties_k }
                set_variable = { global.MIL_cae_war_GUI_CAE_casualties = MIL_cae_war_GUI_temp_value }

                set_temp_variable = { MIL_cae_war_GUI_temp_value = MIL.casualties_k }
                set_variable = { global.MIL_cae_war_GUI_MIL_casualties = MIL_cae_war_GUI_temp_value }

                # 凯撒占据省份：统计凯撒当前控制的千年核心省份数量。
                set_temp_variable = { MIL_cae_war_GUI_temp_value = 0 }
                CAE = {
                    every_controlled_state = {
                        limit = { is_core_of = MIL }
                        add_to_temp_variable = { MIL_cae_war_GUI_temp_value = 1 }
                    }
                }
                set_variable = { global.MIL_cae_war_GUI_CAE_occupied_state_count = MIL_cae_war_GUI_temp_value }

                # 千年占据省份：统计千年当前控制的凯撒核心省份数量。
                set_temp_variable = { MIL_cae_war_GUI_temp_value = 0 }
                MIL = {
                    every_controlled_state = {
                        limit = { is_core_of = CAE }
                        add_to_temp_variable = { MIL_cae_war_GUI_temp_value = 1 }
                    }
                }
                set_variable = { global.MIL_cae_war_GUI_MIL_occupied_state_count = MIL_cae_war_GUI_temp_value }

                # 凯撒剩余核心：统计凯撒当前仍控制的凯撒核心省份数量。
                set_temp_variable = { MIL_cae_war_GUI_temp_value = 0 }
                CAE = {
                    every_controlled_state = {
                        limit = { is_core_of = CAE }
                        add_to_temp_variable = { MIL_cae_war_GUI_temp_value = 1 }
                    }
                }
                set_variable = { global.MIL_cae_war_GUI_CAE_remaining_core_state_count = MIL_cae_war_GUI_temp_value }

                # 千年剩余核心：统计千年当前仍控制的千年核心省份数量。
                set_temp_variable = { MIL_cae_war_GUI_temp_value = 0 }
                MIL = {
                    every_controlled_state = {
                        limit = { is_core_of = MIL }
                        add_to_temp_variable = { MIL_cae_war_GUI_temp_value = 1 }
                    }
                }
                set_variable = { global.MIL_cae_war_GUI_MIL_remaining_core_state_count = MIL_cae_war_GUI_temp_value }

                # =========================================
                # 规则1-3预测：快照偏移量
                # =========================================
                set_temp_variable = { add_eco = 0 }

                # 规则1预测：凯撒每控制一个千年省份，经济状况+2%
                CAE = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = MIL
                        }
                        add_to_temp_variable = { add_eco = 2 }
                    }
                }

                # 规则2预测：凯撒累计每损失7.5k人力，经济状况-1%
                set_temp_variable = { death_population_cae = CAE.casualties_k }
                divide_temp_variable = { death_population_cae = 7.5 }
                subtract_from_temp_variable = { add_eco = death_population_cae }

                # 规则3预测：千年累计每损失2k人力，凯撒经济状况+1%
                set_temp_variable = { death_population_mil = MIL.casualties_k }
                divide_temp_variable = { death_population_mil = 2 }
                add_to_temp_variable = { add_eco = death_population_mil }

                # =========================================
                # 规则4-7预测：每月累积变化量
                # =========================================
                set_temp_variable = { predicted_change = 0 }

                # 规则4预测：凯撒每控制n个千年省份，每月经济状况+0.5n%
                CAE = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = MIL
                        }
                        add_to_temp_variable = { predicted_change = 0.5 }
                    }
                }

                # 规则5预测：凯撒每比千年多出1k服役人口，每月经济状况+0.1%
                set_temp_variable = { manpower_advantage = 0 }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_army_manpower_k }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_navy_manpower_k }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_airforce_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_army_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_navy_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_airforce_manpower_k }
                clamp_temp_variable = {
                    var = manpower_advantage
                    min = 0
                }
                divide_temp_variable = { manpower_advantage = 10 }
                add_to_temp_variable = { predicted_change = manpower_advantage }

                # 规则6预测：凯撒每拥有10k*n个服役人口，每月经济状况-0.1n%
                set_temp_variable = { manpower_cost = 0 }
                add_to_temp_variable = { manpower_cost = CAE.deployed_army_manpower_k }
                add_to_temp_variable = { manpower_cost = CAE.deployed_navy_manpower_k }
                add_to_temp_variable = { manpower_cost = CAE.deployed_airforce_manpower_k }
                divide_temp_variable = { manpower_cost = 100 }
                subtract_from_temp_variable = { predicted_change = manpower_cost }

                # 规则7预测：凯撒每失去一个核心省份（被千年控制），每月经济状况-10%
                MIL = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = CAE
                        }
                        subtract_from_temp_variable = { predicted_change = 10 }
                    }
                }

                # 千年决议预测：抵制/全面禁用凯撒商品带来的每月扣减
                if = {
                    limit = {
                        MIL = { has_country_flag = MIL_cae_eco_weekly_minus_1 }
                    }
                    subtract_from_temp_variable = { predicted_change = 1 }
                }
                if = {
                    limit = {
                        MIL = { has_country_flag = MIL_cae_eco_weekly_minus_1_5 }
                    }
                    subtract_from_temp_variable = { predicted_change = 1.5 }
                }

                # 计算规则1-3本月偏移量变化 = 当前偏移量 - 上月偏移量
                set_temp_variable = { add_eco_change = add_eco }
                subtract_from_temp_variable = { add_eco_change = global.cae_last_add_eco }
                # 合并规则1-3变化量和规则4-7变化量，存入全局变量（仅预测，不结算）
                add_to_temp_variable = { predicted_change = add_eco_change }
                set_variable = { global.cae_predicted_monthly_change = predicted_change }

                # 经济状况竖向进度条：最终经济状况0~200换算为进度条0~100帧。
                set_temp_variable = { cae_economic_progress_frame = global.cae_final_economic_situation }
                clamp_temp_variable = { var = cae_economic_progress_frame min = 0 max = 200 }
                divide_temp_variable = { cae_economic_progress_frame = 2 }
                set_variable = { global.cae_economic_progress_frame = cae_economic_progress_frame }
            }
        }
    }

    # =====================================================
    # 千年-凯撒战争每月经济调整（合并所有规则）
    # 规则1-3：快照偏移量，每次重新计算
    # 规则1：凯撒每控制一个千年省份，经济状况+2%
    # 规则2：凯撒累计每损失7.5k人力，经济状况-1%
    # 规则3：千年累计每损失2k人力，凯撒经济状况+1%
    # 规则4-7：每月累积变化，直接修改基础经济状况
    # 规则4：凯撒每控制n个千年省份，每月经济状况+0.5n%
    # 规则5：凯撒每比千年多出1k服役人口，每月经济状况+0.1%
    # 规则6：凯撒每拥有10k*n个服役人口，每月经济状况-0.1n%
    # 规则7：凯撒每失去一个核心省份（被千年控制），每月经济状况-10%
    # =====================================================
    on_monthly_CAE = {
        effect = {
            if = {
                limit = {
                    has_country_flag = CAE_MIL_war_started
                }

                # =========================================
                # 规则1-3：快照偏移量（基于累积数据，每月重新计算）
                # =========================================
                set_temp_variable = { add_eco = 0 }

                # 规则1：凯撒每控制一个千年省份，经济状况+2%
                CAE = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = MIL
                        }
                        add_to_temp_variable = { add_eco = 2 }
                    }
                }

                # 规则2：凯撒累计每损失7.5k人力，经济状况-1%
                set_temp_variable = { death_population_cae = CAE.casualties_k }
                divide_temp_variable = { death_population_cae = 7.5 }
                subtract_from_temp_variable = { add_eco = death_population_cae }

                # 规则3：千年累计每损失2k人力，凯撒经济状况+1%
                set_temp_variable = { death_population_mil = MIL.casualties_k }
                divide_temp_variable = { death_population_mil = 2 }
                add_to_temp_variable = { add_eco = death_population_mil }

                # =========================================
                # 规则4-7：每月累积变化，直接修改基础经济状况
                # =========================================
                set_temp_variable = { monthly_change = 0 }

                # 规则4：凯撒每控制n个千年省份，每月经济状况+0.5n%
                CAE = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = MIL
                        }
                        add_to_temp_variable = { monthly_change = 0.5 }
                    }
                }

                # 规则5：凯撒每比千年多出1k服役人口，每月经济状况+0.1%
                # 计算公式：(凯撒服役人口 - 千年服役人口) / 10
                set_temp_variable = { manpower_advantage = 0 }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_army_manpower_k }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_navy_manpower_k }
                add_to_temp_variable = { manpower_advantage = CAE.deployed_airforce_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_army_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_navy_manpower_k }
                subtract_from_temp_variable = { manpower_advantage = MIL.deployed_airforce_manpower_k }
                clamp_temp_variable = {
                    var = manpower_advantage
                    min = 0
                }
                divide_temp_variable = { manpower_advantage = 10 }
                add_to_temp_variable = { monthly_change = manpower_advantage }

                # 规则6：凯撒每拥有10k*n个服役人口，每月经济状况-0.1n%
                # 计算公式：凯撒总服役人口 / 100
                set_temp_variable = { manpower_cost = 0 }
                add_to_temp_variable = { manpower_cost = CAE.deployed_army_manpower_k }
                add_to_temp_variable = { manpower_cost = CAE.deployed_navy_manpower_k }
                add_to_temp_variable = { manpower_cost = CAE.deployed_airforce_manpower_k }
                divide_temp_variable = { manpower_cost = 100 }
                subtract_from_temp_variable = { monthly_change = manpower_cost }

                # 规则7：凯撒每失去一个核心省份（被千年控制），每月经济状况-10%
                MIL = {
                    every_controlled_state = {
                        limit = {
                            is_core_of = CAE
                        }
                        subtract_from_temp_variable = { monthly_change = 10 }
                    }
                }

                # 千年决议：抵制/全面禁用凯撒商品带来的每月扣减
                # 仍沿用旧 flag 名，避免重命名影响已有决议状态。
                if = {
                    limit = {
                        MIL = { has_country_flag = MIL_cae_eco_weekly_minus_1 }
                    }
                    subtract_from_temp_variable = { monthly_change = 1 }
                }
                if = {
                    limit = {
                        MIL = { has_country_flag = MIL_cae_eco_weekly_minus_1_5 }
                    }
                    subtract_from_temp_variable = { monthly_change = 1.5 }
                }

                # 将每月变化累积到基础经济状况
                add_to_variable = { global.cae_economic_situation = monthly_change }

                # =========================================
                # 计算最终经济状况 = 基础值 + 快照偏移量
                # =========================================
                set_variable = { global.cae_final_economic_situation = global.cae_economic_situation }
                set_variable = { global.cae_add_economic_situation = add_eco }
                add_to_variable = { global.cae_final_economic_situation = global.cae_add_economic_situation }

                # 经济状况竖向进度条：最终经济状况0~200换算为进度条0~100帧。
                set_temp_variable = { cae_economic_progress_frame = global.cae_final_economic_situation }
                clamp_temp_variable = { var = cae_economic_progress_frame min = 0 max = 200 }
                divide_temp_variable = { cae_economic_progress_frame = 2 }
                set_variable = { global.cae_economic_progress_frame = cae_economic_progress_frame }

                # =========================================
                # 战斗力影响计算
                # 当经济状况高于/低于100时，每偏离2点=1%战斗力影响
                # 影响项：部队攻防、组织度上限及恢复速度、空军效率
                # 当经济状况低于100时，每降低2点，新招募部队初始装备值-1
                # =========================================
                set_temp_variable = { MIL_cae_war_temp_calc = global.cae_final_economic_situation }
                subtract_from_temp_variable = { MIL_cae_war_temp_calc = 100 }
                divide_temp_variable = { MIL_cae_war_temp_calc = 200 }

                set_variable = { global.MIL_cae_war_cae_army_attack_factor = MIL_cae_war_temp_calc }
                set_variable = { global.MIL_cae_war_cae_army_defence_factor = MIL_cae_war_temp_calc }
                set_variable = { global.MIL_cae_war_cae_army_org_factor = MIL_cae_war_temp_calc }
                set_variable = { global.MIL_cae_war_cae_army_org_regain = MIL_cae_war_temp_calc }
                set_variable = { global.MIL_cae_war_cae_air_mission_efficiency = MIL_cae_war_temp_calc }

                # 记录本月偏移量，供下月 daily 计算变化量使用
                set_variable = { global.cae_last_add_eco = add_eco }

                # =========================================
                # 招募部队初始装备值计算
                # 默认值1.0（100%），经济状况低于100时每降低2点减少1%
                # =========================================
                set_temp_variable = { equip_penalty = 100 }
                subtract_from_temp_variable = { equip_penalty = global.cae_final_economic_situation }
                clamp_temp_variable = { var = equip_penalty min = 0 }
                divide_temp_variable = { equip_penalty = 200 }
                set_temp_variable = { equip_factor = 1 }
                subtract_from_temp_variable = { equip_factor = equip_penalty }
                clamp_temp_variable = { var = equip_factor min = 0 max = 1 }
                set_variable = { global.cae_recruit_equipment_factor = equip_factor }

                # =========================================
                # 凯撒经济崩溃判定
                # 当最终经济状况小于等于0时，凯撒战败
                # =========================================
                if = {
                    limit = {
                        NOT = { check_variable = { global.cae_final_economic_situation > 0 } }
                        NOT = { has_global_flag = CAE_MIL_war_cae_defeated }
                    }
                    set_global_flag = CAE_MIL_war_cae_defeated

                    # 双方白和平（包括参战的凯撒子公司）
                    MIL = {
                        white_peace = CAE
                        white_peace = CAP
                        white_peace = CAB
                        white_peace = CAH
                        white_peace = CAO
                    }

                    # MIL获得民族精神"永远飘扬的光辉旗帜"并依次触发三个战后事件
                    MIL = {
                        add_ideas = MIL_glorious_banner_forever
                        country_event = { id = MIL_cae_war.10 days = 1 }  # 光辉的胜利
                        country_event = { id = MIL_cae_war.11 days = 2 }  # 审判战犯
                        country_event = { id = MIL_cae_war.12 days = 3 }  # 凯撒集团大解体
                    }

                    # CAE获得民族精神"史上最失败的投资"和"崩溃的公司体系"
                    CAE = {
                        add_ideas = CAE_worst_investment_ever
                        add_ideas = CAE_collapsed_corporate_system
                    }

                    # 所有凯撒子公司与凯撒总部解除同盟并独立
                    CAE = {
                        release_puppet = CAP
                        release_puppet = CAB
                        release_puppet = CAH
                        release_puppet = CAO
                    }
                    CAP = { leave_faction = yes }
                    CAB = { leave_faction = yes }
                    CAH = { leave_faction = yes }
                    CAO = { leave_faction = yes }

                    # 战争结束，清除国家旗帜，停止后续每月经济结算
                    CAE = { clr_country_flag = CAE_MIL_war_started }
                    MIL = { clr_country_flag = CAE_MIL_war_started }
                }

                # =========================================
                # 凯撒经济胜利判定
                # 当最终经济状况大于等于200时，凯撒战胜
                # =========================================
                if = {
                    limit = {
                        NOT = { check_variable = { global.cae_final_economic_situation < 200 } }
                        NOT = { has_global_flag = CAE_MIL_war_cae_victory }
                        NOT = { has_global_flag = CAE_MIL_war_cae_defeated }
                    }
                    set_global_flag = CAE_MIL_war_cae_victory

                    # CAE获取在MIL_cae_war.6占领的所有state
                    CAE = {
                        # 马尔斯大道占领区
                        transfer_state = 31
                        transfer_state = 471
                        transfer_state = 32
                        transfer_state = 730
                        transfer_state = 33
                        transfer_state = 731
                        transfer_state = 473
                        transfer_state = 320
                        # 千年大街
                        transfer_state = 733
                        transfer_state = 321
                        transfer_state = 322
                        transfer_state = 732
                        transfer_state = 323
                        # 其他占领区
                        transfer_state = 333
                        transfer_state = 341
                        transfer_state = 331
                        transfer_state = 477
                        transfer_state = 745
                        transfer_state = 746
                        transfer_state = 476
                    }

                    # 双方白和平（包括参战的凯撒子公司）
                    MIL = {
                        white_peace = CAE
                        white_peace = CAP
                        white_peace = CAB
                        white_peace = CAH
                        white_peace = CAO
                    }

                    # 战争结束，清除国家旗帜，停止后续每月经济结算
                    CAE = { clr_country_flag = CAE_MIL_war_started }
                    MIL = { clr_country_flag = CAE_MIL_war_started }
                }
            }
        }
    }

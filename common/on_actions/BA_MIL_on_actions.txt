on_actions = {
    # 千年-凯撒战争每周事件
    # Wiki参考：https://hoi4.paradoxwikis.com/On_actions
    # 用于处理每周减少凯撒经济状况的逻辑
    on_weekly_MIL = {
        effect = {
            # 检查是否是千年
            if = {
                limit = {
                    original_tag = MIL
                }

                # 决议1效果：每周凯撒经济状况-1
                if = {
                    limit = {
                        has_country_flag = MIL_cae_eco_weekly_minus_1
                    }
                    subtract_from_variable = {
                        var = global.cae_economic_situation
                        value = 1
                    }
                }

                # 决议2效果：每周凯撒经济状况-1.5（在-1%基础上）
                if = {
                    limit = {
                        has_country_flag = MIL_cae_eco_weekly_minus_1_5
                    }
                    subtract_from_variable = {
                        var = global.cae_economic_situation
                        value = 1.5
                    }
                }
            }
        }
    }

    # 千年-凯撒战争每日经济调整
    # 每天执行一次省份和人力检查
    on_daily_CAE = {
        effect = {

            set_temp_variable = { add_eco = 0 }
            # 检查占领省份，每个省份提供2%经济状况
            # 这里是every_controlled_state还是every_owned_state ?
            CAE = {
                every_controlled_state = {
                    limit = {
                        is_core_of = MIL
                    }
                    add_to_temp_variable = { add_eco = 2 }
                }
            }
            # 检查损失人力 凯撒每损失7.5k减少1%经济状况
            set_temp_variable = { death_population_cae = CAE.casualties_k }
            divide_temp_variable = { death_population_cae = 7.5 }
            subtract_from_temp_variable = { add_eco = death_population_cae }
            # 检查损失人力 千年每损失2k提高1%经济状况
            set_temp_variable = { death_population_mil = CAE.casualties_k }
            divide_temp_variable = { death_population_mil = 2 }
            add_to_temp_variable = { add_eco = death_population_mil }

            # 计算最终经济状况
            set_variable = { global.cae_final_economic_situation = global.cae_economic_situation }
            set_variable = { global.cae_add_economic_situation = add_eco }
            add_to_variable = { global.cae_final_economic_situation = global.cae_add_economic_situation }

            # 计算动态效果
            # === 战斗力影响计算 ===
            # 计算公式：以100为基准，每偏离2点=1%战斗力影响
            set_temp_variable = {  MIL_cae_war_temp_calc  = global.cae_final_economic_situation }
            subtract_from_temp_variable = { MIL_cae_war_temp_calc  = 100 }
            divide_temp_variable = { MIL_cae_war_temp_calc  = 200 }

            set_variable = { gloabl.MIL_cae_war_cae_army_attack_factor = MIL_cae_war_temp_calc }
            set_variable = { gloabl.MIL_cae_war_cae_army_defence_factor = MIL_cae_war_temp_calc }
            set_variable = { gloabl.MIL_cae_war_cae_army_org_factor = MIL_cae_war_temp_calc }
            set_variable = { gloabl.MIL_cae_war_cae_army_org_regain = MIL_cae_war_temp_calc }
            set_variable = { gloabl.MIL_cae_war_cae_air_mission_efficiency = MIL_cae_war_temp_calc }
        }
    }

    # 千年-凯撒战争每周经济调整
    # 每周执行一次凯撒经济状况的规则4-7
    on_weekly_CAE = {
        effect = {
            # 仅当开战后触发
            limit = {
                has_country_flag = CAE_MIL_war_started
            }
            set_temp_variable = { add_eco = 0 }

            # 开战之后，凯撒每控制n个千年省份，则每周凯撒经济状况增加0.5n%
            CAE = {
                every_controlled_state = {
                    limit = {
                        is_core_of = MIL
                    }
                    add_to_temp_variable = { add_eco = 0.5 }
                }
            }

            # 开战之后，凯撒每比千年多出1k*n个服役人口，则每周凯撒经济状况增加0.1n%
            set_temp_variable = { popluation = 0 }
            add_to_temp_variable = { popluation = CAE.deployed_army_manpower_k }
            add_to_temp_variable = { popluation = CAE.deployed_navy_manpower_k }
            add_to_temp_variable = { popluation = CAE.deployed_airforce_manpower_k }
            subtract_from_temp_variable = { popluation = MIL.deployed_army_manpower_k }
            subtract_from_temp_variable = { popluation = MIL.deployed_navy_manpower_k }
            subtract_from_temp_variable = { popluation = MIL.deployed_army_manpower_k }
            clamp_temp_variable = {
                var = popluation
                min = 0
            }
            divide_temp_variable = { popluation = 10 }
            add_to_temp_variable = { add_eco = popluation }

            # 开战之后，凯撒每拥有10k*n个服役人口，则每周凯撒经济状况下降0.1n%
            set_temp_variable = { popluation_2 = 0 }
            add_to_temp_variable = { popluation_2 = CAE.deployed_army_manpower_k }
            add_to_temp_variable = { popluation_2 = CAE.deployed_navy_manpower_k }
            add_to_temp_variable = { popluation_2 = CAE.deployed_airforce_manpower_k }
            divide_temp_variable = { popluation_2 = 10 }
            subtract_from_temp_variable = { add_eco = population2 }

            # 开战之后，凯撒每失去一个核心省份，则每周凯撒经济状况下降10%

            # 计算最终经济状况
            set_variable = { global.cae_final_economic_situation = global.cae_economic_situation }
            set_variable = { global.cae_add_economic_situation = add_eco }
            add_to_variable = { global.cae_final_economic_situation = global.cae_add_economic_situation }

            # 计算动态效果
            # === 战斗力影响计算 ===
            # 计算公式：以100为基准，每偏离2点=1%战斗力影响
            set_temp_variable = {  MIL_cae_war_temp_calc  = global.cae_final_economic_situation }
            subtract_from_temp_variable = { MIL_cae_war_temp_calc  = 100 }
            divide_temp_variable = { MIL_cae_war_temp_calc  = 200 }

            set_variable = { gloabl.MIL_cae_war_cae_army_attack_factor = MIL_cae_war_temp_calc }
            set_variable = { gloabl.MIL_cae_war_cae_army_defence_factor = MIL_cae_war_temp_calc }
            set_variable = { gloabl.MIL_cae_war_cae_army_org_factor = MIL_cae_war_temp_calc }
            set_variable = { gloabl.MIL_cae_war_cae_army_org_regain = MIL_cae_war_temp_calc }
            set_variable = { gloabl.MIL_cae_war_cae_air_mission_efficiency = MIL_cae_war_temp_calc }
        }
    }
}

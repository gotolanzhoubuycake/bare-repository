##显示屏
defined_text = {
    name = launcher_pad_status_sl
    ##没钥匙
    text = {
        trigger = {
            NOT = {
                check_variable = {
                    LP_key_switch_posistion = 3
                }
            }
        }
        localization_key = plz_insert_key_launcher_pad_status_sl
    }
    ##密码阶段
    text = {
        trigger = {
            ##钥匙就位
            check_variable = {
                LP_key_switch_posistion = 3
            }
            ##没有输错密码
            check_variable = {
                LP_PASSWORD_WRONG = 0
            }
            ##没有输对密码
            check_variable = {
                lp_num_index_complete = 0
            }
        }
        localization_key = keyready_launcher_pad_status_sl
    }
    text = {
        trigger = {
            ##输错密码
            check_variable = {
                LP_PASSWORD_WRONG = 1
            }
        }
        localization_key = error_launcher_pad_status_sl
    }
    ##等待供电阶段
    text = {
        trigger = {
            ##钥匙就位
            check_variable = {
                LP_key_switch_posistion = 3
            }
            ##密码正确
            check_variable = {
                lp_num_index_complete = 1
            }
            ##供电没有
            OR = {
                check_variable = {
                    LP_switch_1_posistion = 0
                }
                check_variable = {
                    LP_switch_2_posistion = 0
                }
                check_variable = {
                    LP_switch_3_posistion = 0
                }
            }
        }
        localization_key = password_ready_launcher_pad_status_sl
    }

    ##已供电
    text = {
        trigger = {
            ##钥匙就位
            check_variable = {
                LP_key_switch_posistion = 3
            }
            ##密码正确
            check_variable = {
                lp_num_index_complete = 1
            }
            ##供电就位
            check_variable = {
                LP_switch_1_posistion = 2
            }
            check_variable = {
                LP_switch_2_posistion = 2
            }
            check_variable = {
                LP_switch_3_posistion = 2
            }
        }
        localization_key = power_ready_launcher_pad_status_sl
    }
}

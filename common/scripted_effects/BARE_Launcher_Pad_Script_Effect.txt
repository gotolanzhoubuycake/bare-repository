		limit = {
			check_variable = {
				lp_num_index_00 = 9
			}
			check_variable = {
				lp_num_index_01 = 1
			}
			check_variable = {
				lp_num_index_02 = 7
			}
			check_variable = {
				lp_num_index_03 = 2
			}
			check_variable = {
				lp_num_index_04 = 5
			}
			check_variable = {
				LP_PASSWORD_WRONG = 0
			}
		}
		##停用密码输入系统，因为已经进入下一个阶段
		set_variable = {
			lp_num_index_complete = 1
		}
	}
	##还原密码错误状态
	else_if = {
		limit = {
			check_variable = {
				LP_PASSWORD_WRONG = 1
			}
		}	
		##重置密码输入系统##
		set_variable = {
			lp_num_index = 0
		}
        set_variable = {
			LP_PASSWORD_WRONG = 0
		}
		##还原所有数字
		set_variable = {
			lp_num_index_00 = 0
		}
		set_variable = {
			lp_num_index_01 = 0
		}
		set_variable = {
			lp_num_index_02 = 0
		}
		set_variable = {
			lp_num_index_03 = 0
		}
		set_variable = {
			lp_num_index_04 = 0
		}
	}
	##密码错误
	else = {
		##在屏幕显示密码错误
		set_variable = {
			LP_PASSWORD_WRONG = 1
		}
	}
}

BARE_MIL_Sport_Monthly_effect = {
	#每月支持度更新
	add_to_variable = {
		BARE_MIL_sport_Student_Support_Var = modifier@MIL_sport_student_support_monthly
	}
	BARE_MIL_Sport_Student_Support_Var_Chance_effect = yes

	set_variable = {
		BARE_MIL_sport_Fee_All_Var = modifier@MIL_sport_student_Fee_monthly
	}	
	#经费更新
	set_variable = { #本月有经费
		BARE_MIL_sport_Fee_Have_Var = BARE_MIL_sport_Fee_All_Var
	}
	add_to_variable = { #额外经费
		BARE_MIL_sport_Fee_Have_Var = BARE_MIL_sport_Fee_Extra_Var
	}
	subtract_from_variable = { #每月消耗
		BARE_MIL_sport_Fee_Have_Var = BARE_MIL_sport_Fee_Month_Cost_Var
	}
	#清理限制flag
	clr_country_flag = BARE_MIL_sport_Security_Have_Change
	clr_country_flag = BARE_MIL_sport_Tech_Have_Change
}

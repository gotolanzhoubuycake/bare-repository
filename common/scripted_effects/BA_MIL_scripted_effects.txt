		set_variable = { MIL_award_game_development_completed_count = 0 }
		set_variable = { MIL_award_game_development_political_power = 0 }
		set_variable = { MIL_award_game_development_stability = 0 }
		set_variable = { MIL_award_game_development_annual_great_success = 0 }
		set_variable = { MIL_award_game_development_annual_success = 0 }
		set_variable = { MIL_award_game_development_annual_neutral = 0 }
	}
	if = {
		limit = {
			NOT = { has_active_mission = MIL_award_game_department_new_invention }
		}
		activate_mission = MIL_award_game_department_new_invention
		if = {
			limit = {
				check_variable = { MIL_award_game_development_completed_count > 1 }
			}
			set_temp_variable = {
				MIL_award_game_development_extra_days = MIL_award_game_development_completed_count
			}
			add_to_temp_variable = {
				MIL_award_game_development_extra_days = -1
			}
			multiply_temp_variable = {
				MIL_award_game_development_extra_days = 30
			}
			add_days_mission_timeout = {
				mission = MIL_award_game_department_new_invention
				days = var:MIL_award_game_development_extra_days
			}
			clear_variable = MIL_award_game_development_extra_days
		}
	}
}

BARE_MIL_award_game_development_improve_chance_effect = {
	# 第三次开发起：通常令大受欢迎+5%、多半好评+10%；不温不火归零后改为大受欢迎+10%。
	if = {
		limit = {
			check_variable = { MIL_award_game_development_min < 1 }
		}
		add_to_variable = { MIL_award_game_development_max = -10 }
	}
	else = {
		add_to_variable = { MIL_award_game_development_max = -5 }
		add_to_variable = { MIL_award_game_development_min = -15 }
	}
	clamp_variable = {
		var = MIL_award_game_development_min
		min = 0
		max = 90
	}
	clamp_variable = {
		var = MIL_award_game_development_max
		min = 0
		max = 100
	}
}

BARE_MIL_award_game_development_apply_bonus_effect = {
	if = {
		limit = { has_active_mission = MIL_award_game_development_bonus_timeout }
		remove_mission = MIL_award_game_development_bonus_timeout
	}
	if = {
		limit = {
			NOT = { has_dynamic_modifier = { modifier = MIL_Game_Development_Department_Game_Development } }
		}
		add_dynamic_modifier = { modifier = MIL_Game_Development_Department_Game_Development }
	}
	activate_mission = MIL_award_game_development_bonus_timeout
	force_update_dynamic_modifier = yes
}

BARE_MIL_award_game_development_finish_effect = {
	# 第一次年度粪作暂不结算事件或奖励，直接进入 180 日第二轮；第二次必定大受欢迎；第三次起按概率结算并自动循环。
	if = {
		limit = {
			check_variable = { MIL_award_game_development_completed_count < 1 }
		}
		add_to_variable = { MIL_award_game_development_completed_count = 1 }
		BARE_MIL_award_game_development_start_project_effect = yes
	}
	else_if = {
		limit = {
			check_variable = { MIL_award_game_development_completed_count < 2 }
		}
		add_political_power = 200
		set_variable = { MIL_award_game_development_political_power = 0.20 }
		set_variable = { MIL_award_game_development_stability = 0.05 }
		BARE_MIL_award_game_development_apply_bonus_effect = yes
		add_to_variable = { MIL_award_game_development_annual_great_success = 1 }
		add_to_variable = { MIL_award_game_development_completed_count = 1 }
		set_variable = { MIL_award_game_development_max = 100 }
		set_variable = { MIL_award_game_development_min = 90 }
		country_event = { id = MIL_award_game.10 }
		country_event = { id = MIL_award_game.9 }
		BARE_MIL_award_game_development_start_project_effect = yes
	}
	else = {
		randomize_variable = {
			var = MIL_award_game_development_prob
			distribution = uniform
			min = 0
			max = 100
		}
		if = {
			limit = {
				check_variable = { MIL_award_game_development_prob > MIL_award_game_development_max }
			}
			add_political_power = 200
			set_variable = { MIL_award_game_development_political_power = 0.20 }
			set_variable = { MIL_award_game_development_stability = 0.05 }
			add_to_variable = { MIL_award_game_development_annual_great_success = 1 }
			country_event = { id = MIL_award_game.10 }
		}
		else_if = {
			limit = {
				check_variable = { MIL_award_game_development_prob > MIL_award_game_development_min }
			}
			add_political_power = 150
			set_variable = { MIL_award_game_development_political_power = 0.10 }
			set_variable = { MIL_award_game_development_stability = 0 }
			add_to_variable = { MIL_award_game_development_annual_success = 1 }
			country_event = { id = MIL_award_game.11 }
		}
		else = {
			add_political_power = 100
			set_variable = { MIL_award_game_development_political_power = 0.05 }
			set_variable = { MIL_award_game_development_stability = 0 }
			add_to_variable = { MIL_award_game_development_annual_neutral = 1 }
			country_event = { id = MIL_award_game.12 }
		}
		BARE_MIL_award_game_development_apply_bonus_effect = yes
		add_to_variable = { MIL_award_game_development_completed_count = 1 }
		set_variable = { MIL_award_game_development_max = 100 }
		set_variable = { MIL_award_game_development_min = 90 }
		BARE_MIL_award_game_development_start_project_effect = yes
	}
}

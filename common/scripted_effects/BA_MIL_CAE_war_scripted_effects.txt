# 2026-10-08：两种胜利共用停战步骤，避免各结算入口重复维护参战方名单。
BARE_MIL_CAE_war_white_peace_effect = {
    # 入口：由任一胜利效果在国家作用域调用；固定切换至 MIL 执行双方停战。
    # 关键条件：仅对仍与 MIL 交战的凯撒总部和子公司执行 white_peace。
    # 预期结果：终止 MIL 与 CAE/CAP/CAB/CAH/CAO 的战争；奖励、拆分和清理由调用方执行。
    MIL = {
        # 同一场战争中的前一次白和平可能已使其他参战方停战，因此逐个重新检查。
        if = {
            limit = { has_war_with = CAE }
            white_peace = CAE
        }
        if = {
            limit = { has_war_with = CAP }
            white_peace = CAP
        }
        if = {
            limit = { has_war_with = CAB }
            white_peace = CAB
        }
        if = {
            limit = { has_war_with = CAH }
            white_peace = CAH
        }
        if = {
            limit = { has_war_with = CAO }
            white_peace = CAO
        }
    }
}

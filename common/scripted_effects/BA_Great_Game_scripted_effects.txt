###计算实力分数###
# 功能：计算国家在战术演习系统中的综合实力分数
# 调用者：Great_Game_generate_global_chart 效果
# 参数：无
# 返回：设置以下变量
#   - Great_Game_power_score：总实力分数
#   - Great_Game_power_score_from_army：陆军分数
#   - Great_Game_power_score_from_air：空军分数
#   - Great_Game_power_score_from_factories：工业分数
#   - Great_Game_power_score_from_technologies：科研分数
#   - Great_Game_power_score_factor：连胜/连败修正系数
# 说明：
#   - 陆军分数 = (本国+所有属国部署陆军人数K) × 5
#   - 空军分数 = (本国+所有属国部署空军人数K) × 4
#   - 工业分数 = 军工数×2 + 民工数×1
#   - 科研分数 = 已研究科技数×2
#   - 连胜修正：1 + 连胜次数×0.05（上限1.5倍，即最多+0.5）
#   - 连败修正：1 - 连败次数×0.05（下限0.5倍，即最多-0.5）
#   - 最终分数 = (陆军+空军+工业+科研) × 修正系数
###
Great_Game_calculate_power_score = {
	hidden_effect = {
		set_variable = { Great_Game_power_score_from_army = deployed_army_manpower_k }
		set_variable = { Great_Game_power_score_from_air = deployed_airforce_manpower_k }
		every_subject_country = {
			add_to_variable = { PREV.Great_Game_power_score_from_army = deployed_army_manpower_k }
			add_to_variable = { PREV.Great_Game_power_score_from_air = deployed_airforce_manpower_k }
		}
		multiply_variable = { Great_Game_power_score_from_army = 6 }
		multiply_variable = { Great_Game_power_score_from_air = 2 }
		set_variable = { Great_Game_power_score_from_factories = num_of_military_factories }
		multiply_variable = { Great_Game_power_score_from_factories = 2 }
		add_to_variable = { Great_Game_power_score_from_factories = num_of_civilian_factories }
		set_variable = { Great_Game_power_score_from_technologies = num_researched_technologies }
		multiply_variable = { num_researched_technologies = 2 }
		#连胜/连败的修正
		if = {
			limit = { check_variable = { Great_Game_consecutive_wins > 0 } }
			set_variable = { Great_Game_power_score_factor = Great_Game_consecutive_wins }
			multiply_variable = { Great_Game_power_score_factor = 0.05 }
			clamp_variable = {
				var = Great_Game_power_score_factor
				max = 0.5
			}
		}
		else = {
			set_variable = { Great_Game_power_score_factor = Great_Game_consecutive_defeats }
			multiply_variable = { Great_Game_power_score_factor = -0.05 }
			clamp_variable = {
				var = Great_Game_power_score_factor
				min = -0.5
			}
		}
		add_to_variable = { Great_Game_power_score_factor = 1 }

		set_variable = { Great_Game_power_score = Great_Game_power_score_from_army }
		add_to_variable = { Great_Game_power_score = Great_Game_power_score_from_air }
		add_to_variable = { Great_Game_power_score = Great_Game_power_score_from_factories }
		add_to_variable = { Great_Game_power_score = Great_Game_power_score_from_technologies }
		multiply_variable = { Great_Game_power_score = Great_Game_power_score_factor }
	}
}
###生成全局排行榜###
# 功能：根据所有参与国家的实力分数生成排行榜，并设置排名图标
# 调用者：
#   - Great_Game_invade_succeeded 效果（演习成功后更新榜单）
#   - Great_Game_invade_failed 效果（演习失败后更新榜单）
#   - Great_Game_force_end_exercise 效果（强制终止演习后更新榜单）
#   - 游戏初始化和定期更新
# 参数：无（从 global.Great_Game_overlords 数组读取所有参与国家）
# 返回：设置以下全局变量
#   - global.Great_Game_overlords_chart：按实力排序的国家数组
#   - 每个国家设置以下变量：
#     - Great_Game_rank：排名（1、2、3...）
#     - Great_Game_rank_icon_frame：排名图标框架（1=第1名金色，2=第2-3名银色，3=第4-6名铜色，4=第7名及以后普通）
#     - Great_Game_power_score_faction：如果是盟主，则为派系总分（本国+盟友）
# 说明：
#   - 先调用每个国家的 Great_Game_calculate_power_score 计算实力分数
#   - 使用临时数组存储国家和分数，然后按分数从高到低排序
#   - 最后在数组末尾添加两个0防止访问越界（bug修复用）
###
Great_Game_generate_global_chart = {
	hidden_effect = {
		for_each_scope_loop = {
			array = global.Great_Game_overlords
			Great_Game_calculate_power_score = yes
			add_to_temp_array = { global.Great_Game_overlords_temp = THIS }
			add_to_temp_array = { global.Great_Game_overlords_score_temp = Great_Game_power_score }
			if = {
				limit = { has_country_flag = involved_in_Great_Game_as_ally_leader }
				set_variable = { Great_Game_power_score_faction = Great_Game_power_score }
				for_each_loop = {
					array = Great_Game_temp_allies
					value = v
					add_to_variable = {
						Great_Game_power_score_faction = var:v:Great_Game_power_score
					}
				}
			}
		}
		#产生排行榜
		clear_array = global.Great_Game_overlords_chart
		for_each_loop = {
			array = global.Great_Game_overlords
			find_highest_in_array = {
				array = global.Great_Game_overlords_score_temp
				index = i
			}
			add_to_array = { global.Great_Game_overlords_chart = global.Great_Game_overlords_temp^i }
			remove_from_temp_array = {
				array = global.Great_Game_overlords_temp
				index = i
			}
			remove_from_temp_array = {
				array = global.Great_Game_overlords_score_temp
				index = i
			}
		}
		set_temp_variable = { rank = 1 }
		for_each_scope_loop = {
			array = global.Great_Game_overlords_chart
			set_variable = { Great_Game_rank = rank }
			if = {
				limit = { check_variable = { rank = 1 } }
				set_variable = { Great_Game_rank_icon_frame = 1 }
			}
			else_if = {
				limit = { check_variable = { rank < 4 } }
				set_variable = { Great_Game_rank_icon_frame = 2 }
			}
			else_if = {
				limit = { check_variable = { rank < 7 } }
				set_variable = { Great_Game_rank_icon_frame = 3 }
			}
			else = {
				set_variable = { Great_Game_rank_icon_frame = 4 }
			}
			add_to_temp_variable = { rank = 1 }
		}
		#防bug用的
		add_to_array = { global.Great_Game_overlords_chart = 0 }
		add_to_array = { global.Great_Game_overlords_chart = 0 }
	}
}

###邀请盟友加入演习###
# 功能：邀请盟友国家加入正在进行的战术演习
# 调用者：BA_Great_Game_decisions.txt 中的"邀请盟友"决议
# 参数：需要设置以下临时变量
#   - ally：要邀请的盟友国家
#   - leader：演习的发起方（盟主）
#   - enemy：对手方（敌对盟主）
# 返回：无
# 说明：
#   - 将盟友添加到盟主的 Great_Game_temp_allies 数组
#   - 更新盟主的派系总分 Great_Game_power_score_faction
#   - 根据盟主角色设置盟友为进攻方或防守方标记
#   - 同步盟友的演习相关变量（目标、对手等）
#   - 取消盟友与对方派系所有成员的军事通行权（确保正常交战）
###
Great_Game_invite_to_ally = {
	custom_effect_tooltip = {
		localization_key = Great_Game_join_ally_tt
		ALLY = "[?ally.GetNameWithFlag]"
		LEADER = "[?leader.GetNameWithFlag]"
		ENEMY = "[?enemy.GetNameWithFlag]"
	}
	hidden_effect = {
		var:leader = {
			add_to_array = { Great_Game_temp_allies = var:ally }
			add_to_variable = { Great_Game_power_score_faction = var:ally:Great_Game_power_score }
		}
		var:ally = {
			if = {
				limit = {
					var:leader = { has_country_flag = involved_in_Great_Game_as_attacker }
				}
				set_country_flag = involved_in_Great_Game_as_attacker
			}
			else = {
				set_country_flag = involved_in_Great_Game_as_defender
			}
			set_variable = { Great_Game_invader = var:leader:Great_Game_invader }
			set_variable = { Great_Game_invade_target = var:leader:Great_Game_invade_target }
			set_variable = { Great_Game_invade_target_overlord = var:leader:Great_Game_invade_target_overlord }
			set_variable = { Great_Game_enemy = var:leader:Great_Game_enemy }
			set_variable = { Great_Game_ally_leader = var:leader }
			
		}
		var:enemy = {
			diplomatic_relation = {
				country = var:ally
				relation = offer_military_access
				active = no
			}
			diplomatic_relation = {
				country = var:ally
				relation = military_access
				active = no
			}
			for_each_scope_loop = {
				array = Great_Game_temp_allies
				diplomatic_relation = {
					country = var:ally
					relation = offer_military_access
					active = no
				}
				diplomatic_relation = {
					country = var:ally
					relation = military_access
					active = no
				}
			}
		}
	}
}

###开始演习（宣战并激活任务）###
# 功能：正式启动战术演习，宣战并激活进攻/防守/盟友任务
# 调用者：BA_Great_Game_decisions.txt 中的"发起演习"决议
# 参数：需要设置以下变量
#   - ROOT：进攻方盟主
#   - FROM：目标托管地区
#   - FROM.OVERLORD：防守方盟主
#   - ROOT.Great_Game_temp_allies：进攻方盟友数组（如有）
#   - FROM.OVERLORD.Great_Game_temp_allies：防守方盟友数组（如有）
# 返回：无
# 说明：
#   - 首先检查防守方是否在免战列表中（global.Great_Game_cannot_invade）
#   - 如果在免战列表：
#     - 显示取消提示
#     - 清理双方所有演习相关变量和标记
#   - 如果不在免战列表：
#     - 进攻方对目标地区宣战（战争类型：take_trusteeship）
#     - 激活进攻方任务 college_exercise_ongoing_attacker
#     - 所有进攻方盟友加入战争并激活 college_exercise_ongoing_ally 任务
#     - 防守方激活 college_exercise_ongoing_defender 任务
#     - 所有防守方盟友加入战争并激活 college_exercise_ongoing_ally 任务
###
Great_Game_start_invasion = {
	hidden_effect = {
		#检查是否在免战列表中
		if = {
			limit = {
				is_in_array = { global.Great_Game_cannot_invade = FROM.OVERLORD }
			}
			#在免战列表中，取消演习
			custom_effect_tooltip = Great_Game_invasion_cancelled_due_to_cease_fire_tt
			#清理变量
			clear_array = Great_Game_temp_allies
			clr_country_flag = involved_in_Great_Game_as_attacker
			clr_country_flag = involved_in_Great_Game_as_ally_leader
			clear_variable = Great_Game_invader
			clear_variable = Great_Game_invade_target
			clear_variable = Great_Game_invade_target_overlord
			clear_variable = Great_Game_enemy
			#防守方也清理
			FROM.OVERLORD = {
				clear_array = Great_Game_temp_allies
				clr_country_flag = involved_in_Great_Game_as_defender
				clr_country_flag = involved_in_Great_Game_as_ally_leader
				clear_variable = Great_Game_invader
				clear_variable = Great_Game_invade_target
				clear_variable = Great_Game_invade_target_overlord
				clear_variable = Great_Game_enemy
			}
		}
		else = {
			#不在免战列表，正常开始演习
			#攻方
			declare_war_on = {
				target = FROM
				type = take_trusteeship
			}
			activate_mission = college_exercise_ongoing_attacker
			#攻方盟友
			for_each_scope_loop = {
				array = Great_Game_temp_allies
				add_to_war = {
					targeted_alliance = ROOT
					enemy = FROM.OVERLORD
				}
				activate_mission = college_exercise_ongoing_ally
			}
			FROM.OVERLORD = {
				#守方
				activate_mission = college_exercise_ongoing_defender
				#守方盟友
				for_each_scope_loop = {
					array = Great_Game_temp_allies
					add_to_war = {
						targeted_alliance = PREV
						enemy = ROOT
					}
					activate_mission = college_exercise_ongoing_ally
				}
			}
		}
	}
}

###演习进攻成功###
# 功能：处理进攻方占领目标地区后的胜利结算
# 调用者：BA_GST_ZhanZheng_actions.txt 中的 on_state_control_changed 事件
# 参数：
#   - ROOT：进攻方盟主
#   - Great_Game_invade_target：目标地区
#   - Great_Game_invade_target_overlord：防守方盟主
#   - Great_Game_enemy：对手方（与防守方盟主相同）
# 返回：无
# 说明：
#   胜利方（进攻方）：
#   - 增加连胜次数，清零连败次数
#   - 调用 Great_Game_get_coins 获得代币奖励
#   - 添加战斗记录（记录目标、对手、结果=1胜利、阵营=1进攻方、日期）
#   - 所有盟友与对手白和退出战争
#   - 清理演习标记和变量
#   - 获得目标地区的托管权（根据资格分配给CEN/CAE/本国）
#   - 进入演习冷却期（180天）
#
#   失败方（防守方）：
#   - 增加连败次数，清零连胜次数
#   - 调用 Great_Game_get_coins 获得少量代币奖励
#   - 添加战斗记录（记录目标、对手、结果=2失败、阵营=2防守方、日期）
#   - 所有盟友与对手白和退出战争
#   - 从托管地区数组中移除该地区
#   - 清理演习标记和变量
#
#   最后更新全局排行榜
###
Great_Game_invade_succeeded = {
	custom_effect_tooltip = Great_Game_invade_succeeded_tt
	Great_Game_get_coins = yes
	hidden_effect = {
		#增加连胜
		add_to_variable = { Great_Game_consecutive_wins = 1 }
		clear_variable = Great_Game_consecutive_defeats
		#记录
		set_variable = { Great_Game_record_target^Great_Game_record_count = var:Great_Game_invade_target }
		set_variable = { Great_Game_record_enemy^Great_Game_record_count = var:Great_Game_enemy }
		set_variable = { Great_Game_record_result^Great_Game_record_count = 1 }
		set_variable = { Great_Game_record_faction^Great_Game_record_count = 1 }
		set_variable = { Great_Game_record_date^Great_Game_record_count = global.date }
		Great_Game_add_record = yes
		#盟友退战
		every_other_country = {
			limit = {
				has_war_together_with = ROOT
				Great_Game_is_involved = yes
			}
			every_enemy_country = {
				limit = { Great_Game_is_involved = yes }
				white_peace = PREV
			}
		}
		#白和
		every_enemy_country = {
			limit = { Great_Game_is_involved = yes }
			white_peace = ROOT
		}
		clear_array = Great_Game_temp_allies
		var:Great_Game_invade_target_overlord = {
			Great_Game_get_coins = yes
			#增加连败
			add_to_variable = { Great_Game_consecutive_defeats = 1 }
			clear_variable = Great_Game_consecutive_wins
			#记录
			set_variable = { Great_Game_record_target^Great_Game_record_count = var:Great_Game_invade_target }
			set_variable = { Great_Game_record_enemy^Great_Game_record_count = ROOT }
			set_variable = { Great_Game_record_result^Great_Game_record_count = 2 }
			set_variable = { Great_Game_record_faction^Great_Game_record_count = 2 }
			set_variable = { Great_Game_record_date^Great_Game_record_count = global.date }
			Great_Game_add_record = yes
			#盟友退战
			every_other_country = {
				limit = {
					has_war_together_with = PREV
					Great_Game_is_involved = yes
				}
				every_enemy_country = {
					limit = { Great_Game_is_involved = yes }
					white_peace = PREV
				}
			}
			clear_array = Great_Game_temp_allies
			remove_from_array = {
				array = BA_trusteeship_areas
				value = var:Great_Game_invade_target
			}
			
			clr_country_flag = involved_in_Great_Game_as_defender
			clr_country_flag = involved_in_Great_Game_as_ally_leader
			clear_variable = Great_Game_invader
			clear_variable = Great_Game_invade_target
			clear_variable = Great_Game_invade_target_overlord
			clear_variable = Great_Game_enemy
		}
	
		#夺取托管地区
		add_to_array = { BA_trusteeship_areas = var:Great_Game_invade_target }
		if = {
			limit = { Great_Game_can_be_CEN_overlord = yes }
			CEN = {
				set_autonomy = {
					target = var:Great_Game_invade_target
					autonomy_state = autonomy_CEN_trusteeship_area
				}
			}
			var:Great_Game_invade_target = {
				if = {
					limit = {
						SRT = {
							#TODO:SRT脱队
							always = no
						}
					}
					add_ideas = trusteeship_of_CEN_SRT_no
				}
				else = {
					add_ideas = trusteeship_of_CEN_SRT_yes
				}
			}
		}
		else_if = {
			limit = { Great_Game_can_be_ceaser_overlord = yes }
			CAE = {
				set_autonomy = {
					target = var:Great_Game_invade_target
					autonomy_state = autonomy_Ceaser_trusteeship_area
				}
			}
			var:Great_Game_invade_target = {
				add_ideas = trusteeship_of_CAE
			}
		}
		else = {
			set_autonomy = {
				target = var:Great_Game_invade_target
				autonomy_state = autonomy_bare_trusteeship_area
			}
		}
		clr_country_flag = involved_in_Great_Game_as_attacker
		clr_country_flag = involved_in_Great_Game_as_ally_leader
		clear_variable = Great_Game_invader
		clear_variable = Great_Game_invade_target
		clear_variable = Great_Game_invade_target_overlord
		clear_variable = Great_Game_enemy
		set_country_flag = Great_Game_invade_cd
		activate_mission = Great_Game_invade_cd
		Great_Game_generate_global_chart = yes
	}
}

#   - Great_Game_invade_target_overlord：防守方盟主
#   - Great_Game_enemy：对手方（与防守方盟主相同）
# 返回：无
# 说明：
#   胜利方（防守方）：
#   - 增加连胜次数，清零连败次数
#   - 调用 Great_Game_get_coins 获得代币奖励

Great_Game_invade_failed = {
	custom_effect_tooltip = Great_Game_invade_failed_tt
	Great_Game_get_coins = yes
	hidden_effect = {

###获得代币奖励###
# 功能：根据演习中造成的敌方伤亡和战斗表现计算并获得代币奖励
# 调用者：
#   - Great_Game_invade_succeeded 效果（胜败双方各调用）
#   - Great_Game_invade_failed 效果（胜败双方各调用）
#   - Great_Game_ally_leave 效果（退出盟友调用）
#   - Great_Game_force_end_exercise 效果（双方各调用）
# 参数：
#   - 需要先设置 Great_Game_get_coins_from_combat 变量（战斗中累积的击杀得分）
# 返回：
#   - Great_Game_coins 增加计算出的代币数量（上限9999）
#   - Great_Game_get_coins_from_combat 清零
# 说明：
#   代币计算公式：
#   - 基础 = 所有敌方国家的总伤亡人数K × 20
#   - 战斗奖励 = Great_Game_get_coins_from_combat（战斗中累积）
#   - 最终代币 = 基础 + 战斗奖励
#   - 如果是盟主（有 involved_in_Great_Game_as_ally_leader 标记），则伤亡部分 ÷ 2
#   - 代币数量四舍五入取整，最终限制在 0-9999 范围内
###
Great_Game_get_coins = {
	for_each_scope_loop = {
		array = enemies
		if = {
			limit = { Great_Game_is_involved = yes }
			add_to_temp_variable = { total_casualties_k = casualties_k }
		}
	}
	multiply_temp_variable = { total_casualties_k = 20 }
	set_temp_variable = { coins = total_casualties_k }
	add_to_temp_variable = { coins = Great_Game_get_coins_from_combat }
	clear_variable = Great_Game_get_coins_from_combat
	if = {
		limit = { has_country_flag = involved_in_Great_Game_as_ally_leader }
		divide_temp_variable = { total_casualties_k = 2 }
	}
	round_temp_variable = coins
	custom_effect_tooltip = Great_Game_get_coins_tt
	add_to_variable = { Great_Game_coins = coins }
	clamp_variable = {
		var = Great_Game_coins
		min = 0
		max = 9999
	}
}

###全面免战：添加免战状态并终止/取消所有相关演习###
# 功能：使国家进入全面免战状态，终止所有进行中的演习并取消准备中的演习
# 调用者：BA_Great_Game_decisions.txt 中的"全面免战"决议
# 参数：
#   - ROOT：要进入免战状态的国家
# 返回：
#   - 将 ROOT 添加到 global.Great_Game_cannot_invade 数组
# 说明：
#   执行流程：
#   1. 将 ROOT 添加到全局免战列表（如果尚未在列表中）
#
#   2. 终止 ROOT 正在参与的演习：
#      - 如果是盟友：调用 Great_Game_ally_leave 退出演习并结算代币
#      - 如果是进攻方主导：调用 Great_Game_force_end_exercise 终止整个演习
#      - 如果是防守方主导：由进攻方触发 Great_Game_force_end_exercise 终止演习
#
#   3. 清理 ROOT 准备中的演习变量（调用 Great_Game_cancel_preparation）
#
#   4. 检查其他国家是否正在对 ROOT 发起演习（进行中或准备中）：
#      - 遍历所有进攻方，检查其 Great_Game_invade_target_overlord 是否为 ROOT
#      - 如果是盟友：调用 Great_Game_ally_leave 退出
#      - 如果是主导方：调用 Great_Game_force_end_exercise 终止
#
#   5. 清理其他国家针对 ROOT 的准备中演习变量
#
#   免战期间效果：
#   - 无法被其他国家选为演习目标（决议中会过滤掉）
#   - 无法主动发起演习（决议不可见或不可用）
#   - 演习开始时会检查目标是否在免战列表，如果是则自动取消
###
Great_Game_total_cease_fire = {
	custom_effect_tooltip = Great_Game_total_cease_fire_tt
	hidden_effect = {
		#1. 对所有有资格参与演习的势力添加免战
		set_temp_variable = { target = THIS }
		if = {
			limit = {
				NOT = { is_in_array = { global.Great_Game_cannot_invade = target } }
			}
			add_to_array = { global.Great_Game_cannot_invade = target }
		}

		#2. 终止正在进行的演习（区分盟友和主导方）
		if = {
			limit = {
				OR = {
					has_country_flag = involved_in_Great_Game_as_attacker
					has_country_flag = involved_in_Great_Game_as_defender
				}
			}
			#检查是否是盟友（盟友有Great_Game_ally_leader变量）
			if = {
				limit = { has_variable = Great_Game_ally_leader }
				#是盟友，只退出演习并结算代币
				Great_Game_ally_leave = yes
			}
			else = {
				#是主导方，终止整个演习
				if = {
					limit = { has_country_flag = involved_in_Great_Game_as_attacker }
					set_temp_variable = {
						var = Great_Game_total_cease_fire_Target
						value = THIS
					}
					Great_Game_force_end_exercise = yes
				}
				else_if = {
					limit = { has_country_flag = involved_in_Great_Game_as_defender }
					set_temp_variable = {
						var = Great_Game_total_cease_fire_Target
						value = Great_Game_invader
					}
					#由进攻方触发终止
					var:Great_Game_invader = {
						Great_Game_force_end_exercise = yes
					}
				}
			}
		}

		#3. 清理准备中的演习变量
		Great_Game_cancel_preparation = yes

		#4. 检查是否有其他势力正在对ROOT发起演习
		every_country = {
			limit = {
				has_country_flag = involved_in_Great_Game_as_attacker
				has_variable = Great_Game_invade_target_overlord
				check_variable = { Great_Game_invade_target_overlord = PREV }
			}
			#检查是否是盟友
			if = {
				limit = { has_variable = Great_Game_ally_leader }
				#是盟友，只退出演习
				Great_Game_ally_leave = yes
			}
			else = {
				#是主导进攻方，终止整个演习
				Great_Game_force_end_exercise = yes
			}
		}

		#5. 检查针对ROOT的准备中演习（清理变量）
		every_country = {
			limit = {
				has_variable = Great_Game_invade_target_overlord
				check_variable = { Great_Game_invade_target_overlord = PREV }
			}
			Great_Game_cancel_preparation = yes
		}
	}
}

###解除全面免战：从免战列表移除###
# 功能：解除国家的全面免战状态，恢复可以参与演习
# 调用者：BA_Great_Game_decisions.txt 中的"解除全面免战"决议
# 参数：
#   - ROOT：要解除免战的国家
#   - target：临时变量，设置为 THIS（ROOT本身）
# 返回：
#   - 从 global.Great_Game_cannot_invade 数组中移除 ROOT
# 说明：
#   - 检查 ROOT 是否在免战列表中，如果在则移除
#   - 解除后立即生效：
#     - 其他国家可以选择 ROOT 的托管地区作为演习目标
#     - ROOT 可以主动发起演习
#   - 此效果仅移除免战状态，不影响已结束的演习或冷却期
###
Great_Game_lift_total_cease_fire = {
	custom_effect_tooltip = Great_Game_lift_total_cease_fire_tt
	hidden_effect = {
		if = {
			limit = {
				is_in_array = { global.Great_Game_cannot_invade = THIS }
			}
			remove_from_array = { global.Great_Game_cannot_invade = THIS }
		}
	}
}

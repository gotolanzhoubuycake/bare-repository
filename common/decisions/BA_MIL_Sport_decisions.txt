# 1. 结构参考原版苏联宣传鼓动部 `SOV_propaganda_campaigns`，保留“持续时间 / 冷却 / 并发上限 / 单项递增成本”的骨架。
# 2. 用户要求当前先不要使用专用决议 GUI，因此这里把宣传活动写成普通文本决议，但仍复用通用宣传系统变量。
# 3. 用户还要求把脚本放进千年的决议文件里统一维护，因此虽然类别名仍为全局宣传活动，首批千年宣传决议直接定义在本文件。
# 4. `visible` 负责把这组决议严格限定在千年与晄轮大祭筹备阶段，避免其他势力误见。
# 5. `available` 只检查“是否还有宣传栏位”，把宣传系统的并发控制集中交给通用 trigger，后续扩展时不容易写散。
# 6. `complete_effect` 负责立即结算本次宣传的即时收益，并登记一个正在运行的宣传活动。
# 7. `remove_effect` 在持续时间结束后负责释放栏位，并把该条宣传的下次成本按全局倍率抬高。

			political_power_factor = 0.05
		}

		complete_effect = {
			# 登记一个进行中的宣传活动，并累计系统总发动次数。
			add_to_variable = {
				BARE_propaganda_active_campaigns_var = 1
			}
			add_to_variable = {
				BARE_propaganda_campaigns_total_var = 1
			}

			# 即时收益：
			# 这类宣传的核心回报是“快速稳住校内舆论”，因此学生支持度提升幅度高于综合评分。
			add_to_variable = {
				BARE_MIL_sport_Student_Support_Var = 10
			}
			add_to_variable = {
				BARE_MIL_sport_Score_Var = 5
			}
			BARE_MIL_Sport_Student_Support_Var_Chance_effect = yes
			custom_effect_tooltip = MIL_Public_Pull_Together_tt
		}

		remove_effect = {
			# 活动到期后释放一个宣传栏位。
			add_to_variable = {
				BARE_propaganda_active_campaigns_var = -1
			}

			# 递增成本：
			# 复用通用倍率逻辑，这样后面别的学院接入时不需要重写“涨价规则”。
			set_variable = {
				BARE_propaganda_cost_var = BARE_MIL_public_pull_together_cost_var
			}
			BARE_increase_propaganda_campaign_cost_effect = yes
			set_variable = {
				BARE_MIL_public_pull_together_cost_var = BARE_propaganda_cost_var
			}
			clear_variable = BARE_propaganda_cost_var
			custom_effect_tooltip = BARE_propaganda_campaign_cooldown_tt
		}
	}
